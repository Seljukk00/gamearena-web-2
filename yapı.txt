JS Wrap Sistemi
Her mod dosyası ortak showScreen ve handleMessage fonksiyonlarını wrap ediyor:

JavaScript

const _prevShowScreen = showScreen;
showScreen = function(screenName) {
    _prevShowScreen(screenName);
    // Modun kendi ekranlarını gizle/göster
};

const _prevHandleMessage = handleMessage;
handleMessage = function(msg) {
    if (msg.type.startsWith("ml_")) { /* handle */ return; }
    _prevHandleMessage(msg);
};
